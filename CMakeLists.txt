cmake_minimum_required(VERSION 3.22)
project(DesktopTasks VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
include(CTest)
find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets Network Test)
if(WIN32)
    include(FetchContent)
    set(BUILD_WITH_QT6 ON CACHE BOOL "" FORCE)
    set(BUILD_TRANSLATIONS OFF CACHE BOOL "" FORCE)
    set(BUILD_TESTING_SAVED ${BUILD_TESTING})
    set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(qtkeychain
        URL https://codeload.github.com/frankosterfeld/qtkeychain/tar.gz/ad7344c45a86a4f66cbafc4b081b5f7b876cb0b7
        URL_HASH SHA256=80774d512a4663093518c2b0185ef2676b067a0a8b7f8a4fbeab475631276d7b)
    FetchContent_MakeAvailable(qtkeychain)
    set(BUILD_TESTING ${BUILD_TESTING_SAVED} CACHE BOOL "" FORCE)
else()
    find_package(Qt6Keychain REQUIRED)
    find_package(LayerShellQt QUIET)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(XCB REQUIRED IMPORTED_TARGET xcb)
endif()
add_library(tasks_core STATIC Source/Store.cpp Source/Api.cpp Source/Auth.cpp)
target_include_directories(tasks_core PUBLIC Source)
target_link_libraries(tasks_core PUBLIC Qt6::Network qt6keychain)
add_library(tasks_ui STATIC Source/Widget.cpp Source/Desktop.cpp)
target_link_libraries(tasks_ui PUBLIC tasks_core Qt6::Widgets)
add_executable(desktop-tasks WIN32 Source/main.cpp)
target_link_libraries(desktop-tasks PRIVATE tasks_ui)
if(WIN32)
    target_link_libraries(tasks_ui PRIVATE user32)
elseif(UNIX)
    target_link_libraries(tasks_ui PRIVATE PkgConfig::XCB)
    if(LayerShellQt_FOUND)
        target_link_libraries(tasks_ui PRIVATE LayerShellQt::Interface)
        target_compile_definitions(tasks_ui PRIVATE HAVE_LAYER_SHELL)
    endif()
endif()
if(MSVC)
    target_compile_options(tasks_core PRIVATE /W4 /utf-8)
    target_compile_options(tasks_ui PRIVATE /W4 /utf-8)
    target_compile_options(desktop-tasks PRIVATE /W4 /utf-8)
else()
    target_compile_options(tasks_core PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(tasks_ui PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(desktop-tasks PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
    add_executable(tasks_tests tests/Tests.cpp)
    target_link_libraries(tasks_tests PRIVATE tasks_core Qt6::Test)
    add_test(NAME tasks COMMAND tasks_tests)
    add_executable(widget_tests tests/WidgetTests.cpp)
    target_link_libraries(widget_tests PRIVATE tasks_ui Qt6::Test)
    add_test(NAME widget COMMAND widget_tests)
    set_tests_properties(tasks widget PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
install(TARGETS desktop-tasks RUNTIME DESTINATION bin)
install(FILES packaging/desktop-tasks.desktop DESTINATION share/applications)
install(FILES LICENSE README.md THIRD_PARTY_NOTICES.md DESTINATION share/doc/desktop-tasks)
install(DIRECTORY docs DESTINATION share/doc/desktop-tasks)
set(CPACK_GENERATOR DEB)
set(CPACK_PACKAGE_NAME desktop-tasks)
set(CPACK_PACKAGE_CONTACT "https://github.com/alexalmny/desktop-tasks/issues")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Native two-column Google Tasks desktop widget")
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_DEBIAN_PACKAGE_DEPENDS "qt6-wayland, libqt6keychain1, layer-shell-qt")
set(CPACK_PACKAGING_INSTALL_PREFIX /usr)
include(CPack)
